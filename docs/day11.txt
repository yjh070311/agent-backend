1. [[RESTful API 设计]] — 用 HTTP 动词表达资源操作

它是什么： RESTful 把「对资源的操作」映射到「HTTP 方法 + URL」：GET /conversations（列出资源）、POST /conversations（创建）、GET /conversations/{id}（查看单个）、PUT/PATCH /conversations/{id}（更新）、DELETE /conversations/{id}（删除）。资源用名词（复数），动作用 HTTP 方法，不用 POST /createConversation 这种动词 URL。

它解决 Agent 工程里的什么痛点： 你的 API 要被别人（前端、别的 Agent、同事）调用。如果 URL 设计混乱（一会儿 POST /create 一会儿 GET /getAll），调用方得翻遍文档才能用。RESTful 约定让「猜都能猜到接口长什么样」——拿到 /conversations 就知道 GET 是列表、POST 是新建。

今天为什么必须学： 今天把 Day 10 的 ORM 能力暴露成 HTTP 接口，实现会话和消息的完整 CRUD。这是 AI-Interview 主项目「面试记录管理」的雏形——Week 7 的面试报告、Week 8 的历史记录都靠这套 CRUD。

新手最容易踩什么坑：
- 用 POST 做查询（POST /getMessages）、用 GET 做删除——动词用错，缓存和幂等性全乱
- 路径用单数 GET /conversation/1，RESTful 惯例是复数 /conversations/1
- 分页信息塞在响应 body 里乱放，没有统一的格式

---

2. [[分页]] + 统一响应 — 让 API 可控、可预期

它是什么： 分页用 limit（每页条数）+ offset（跳过条数）控制返回量，响应里带 total（总数）让调用方知道还有多少。统一响应把每个接口的返回包成固定结构 {code, message, data}——成功和失败都是这个外壳，调用方写一次解析逻辑就够。

它解决 Agent 工程里的什么痛点： 一个会话可能有几百条消息，GET /messages 全量返回会撑爆内存和带宽。分页让客户端「一页页拉」。统一响应解决「各接口返回格式不一」——如果列表接口返回裸数组、详情接口返回裸对象、错误接口返回另一套，调用方要写 N 套解析。

今天为什么必须学： 今天要给 CRUD 接口加分页，并引入统一响应格式（这是 Day 12 统一异常处理的铺垫）。这两个习惯贯穿 Week 3-8 的所有接口。

新手最容易踩什么坑：
- 只返回 data 不返回 total，客户端不知道还有没有下一页
- 分页参数不设默认值和上限——limit=1000000 一次拉爆数据库
- 统一响应只在成功时用，错误时又变成裸 JSON——破坏一致性

app/utils/response.py

"""
统一响应封装。

所有接口的成功和失败响应都走这个外壳：
    {"code": 0, "message": "success", "data": {...}}

统一响应的价值：
    1. 调用方写一次解析逻辑，适配所有接口
    2. 业务状态码（code）和 HTTP 状态码（status
约定：
    code=0 表示成功，非 0 表示业务错误
    data 放实际返回数据（成功时）
    message 放人类可读的说明
"""

from typing import Any, Generic, TypeVar, Optional

from pydantic import BaseModel

# 泛型变量——让 ApiResponse 的 data 字段类型可参数化
T = TypeVar("T")


class ApiResponse(BaseModel, Generic[T]):
    """统一 API 响应外壳。

    Generic[T] 让 data 的类型可以变化：
        ApiResponse[ConversationOut] 表示 data 是 ConversationOut
        ApiResponse[list[ConversationOut]] 表示 data 是列表
    """

    code: int = 0                    # 0=成功，非0=业务错误码
    message: str = "success"         # 人类可读的说明
    data: Optional[T] = None         # 实际数据，成功时有值

    @classmethod
    def success(cls, data: Any = None, message: str = "success") -> "ApiResponse":
        """构造成功响应。"""
        return cls(code=0, message=message, data=data)

    @classmethod
    def error(cls, code: int, message: str) -> "ApiResponse":
        """构造业务错误响应。"""
        return cls(code=code, message=message, data=None)


class PaginatedData(BaseModel, Generic[T]):
    """分页数据结构——嵌套在 ApiResponse.data 里。

    items 是当前页的数据，total 是符合条件的总条数，
    limit/offset 回显本次分页参数，方便调用方调试。
    """

    items: list[T]                   # 当前页数据
    total: int                       # 总条数（用于计算总页数）
    limit: int                       # 每页条数
    offset: int                      # 偏移量

"""会话接口的 Pydantic 模型。"""

from datetime import datetime

from pydantic import BaseModel, Field


class ConversationCreate(BaseModel):
    """创建会话的请求体。"""
    title: str = Field(
        default="新对话",
        min_length=1,
        max_length=100,
        description="会话标题",
    )


class ConversationUpdate(BaseModel):
    """更新会话的请求体（字段都可选，只更新传入的字段）。"""
    title: str | None = Field(
        default=None,
        min_length=1,
    )


class ConversationOut(BaseModel):
    """会话的响应体（从 ORM 模型转出的安全输出）。

    注意：不包含 user_id 之外不该暴露的字段。
    """

    id: int
    title: str
    user_id: int
    created_at: datetime

    # 让 Pydantic 能从 ORM 对象直接转换
    # 这是 SQLAlchemy 和 Pydantic 协作的关键配置
    model_config = {"from_attributes": True}

"""
会话 CRUD 路由——完整的增删改查。

接口清单：
    GET    /conversations          → 列出会话（分页）
    POST   /conversations          → 创建会话

    PATCH  /conversations/{id}     → 更新会话标题
    DELETE /conversations/{id}     → 删除会话

RESTful 设计要点：
    - 资源用复数名词 conversations
    - 操作用 HTTP 方法表达（不用 /create、/delete 这种动词路径）
    - {id} 是路径参数，指向具体资源
"""

from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.orm import Session

from app.database import get_db
from app.models import Conversation

from app.utils.response import ApiResponse, PaginatedData

router = APIRouter(prefix="/conversations", tags=["会话"])


@router.get("", response_model=ApiResponse[PaginatedData[ConversationOut]])
async def list_conversations(
    limit: int = Query(default=20, ge=1, le=100),
    offset: int = Query(default=0, ge=0),
    db: Session = Depends(get_db),
):
    """列出会话（分页）。

    Query 参数：
        limit: 每页条数（1-100，默认 20）
        offset: 跳过前 N 条（默认 0）

    返回分页结构：{items, total, limit, offset}
    """
    # 查总数——用于前端计算总页数
    total = db.query(Conversation).count()

    # 查当前页——按创建时间倒序（最新的在前）
    items = (
        db.query(Conversation)
        .order_by(Conversation.created_at.desc())
        .offset(offset)
        .limit(limit)
        .all()
    )

    return ApiResponse.success(
        data=PaginatedData(
            items=items,  # Pydantic 会通过 from_attributes 自动转换 ORM 对象
            total=total,
            limit=limit,
            offset=offset,
        )
    )


@router.post("", response_model=ApiResponse[ConversationOut], status_code=201)
async def create_conversation(
    payload: ConversationCreate,
    db: Session = Depends(get_db),
):
    """创建会话。

    status_code=201：RESTful 约定「创建成功返回 201 Created」，
    而不是默认的 200。
    """
    # 注意：Day 11 先写死 user_id=1（因为还没有认证系统）
    # Day 13（JWT）会改成从当前登录用户取 user_id
    conversation = Conversation(
        user_id=1,
        title=payload.title,
    )
    db.add(conversation)
    db.commit()
    db.refresh(conversation)

    return ApiResponse.success(data=conversation)


@router.get("/{conversation_id}", response_model=ApiResponse[ConversationOut])
async def get_conversation(
    conversation_id: int,
    db: Session = Depends(get_db),
):
    """查看单个会话。"""
    conversation = db.query(Conversation).filter(Conversation.id == conversation_id).first()
    if conversation is None:
    return ApiResponse.success(data=conversation)


@router.patch("/{conversation_id}", response_model=ApiResponse[ConversationOut])
async def update_conversation(
    conversation_id: int,
    payload: ConversationUpdate,
    db: Session = Depends(get_db),
):
    """更新会话标题。

    用 PATCH 而非 PUT：PATCH 是「部分更新」，只改传入的字段；
    PUT 是「完整替换」，要传全量字段。
    """
    conversation = db.query(Conversation).filter(Conversation.id == conversation_id).first()
    if conversation is None:
        raise HTTPException(status_code=404, detail="会话不存在")

    # 只更新传入的字段（model_dump 排除 None）
    update_data = payload.model_dump(exclude_unset=True)
    for key, value in update_data.items():
        setattr(conversation, key, value)

    db.commit()
    db.refresh(conversation)
    return ApiResponse.success(data=conversation)


@router.delete("/{conversation_id}", status_code=204)
async def delete_conversation(
    conversation_id: int,
    db: Session = Depends(get_db),
):
    """删除会话（连带级联删除其所有消息）。

    status_code=204：RESTful 约定「删除成功返回 204 No Content」，
    响应体为空——因为资源已经没了，没什么可返回的。
    """
    conversation = db.query(Conversation).filter(Conversation.id == conversation_id).first()
    if conversation is None:
        raise HTTPException(status_code=404, detail="会话不存在")

    db.delete(conversation)
    db.commit()
    # 204 无响应体，直接 return None（FastAPI 会处理）
    return None

"""消息路由——消息的分页查询。"""

from fastapi import APIRouter, Depends, Query
from sqlalchemy.orm import Session

from app.database import get_db
from app.models import Message
from app.schemas.message import MessageOut
from app.utils.response import ApiResponse, PaginatedData

router = APIRouter(prefix="/conversations/{conversation_id}/messages", tags=["消息"])


@router.get("", response_model=ApiResponse[PaginatedData[MessageOut]])
async def list_messages(
    conversation_id: int,
    limit: int = Query(default=20, ge=1, le=100),
    offset: int = Query(default=0, ge=0),
    db: Session = Depends(get_db),
):
    """列出某个会话下的消息（分页）。

    路径设计：嵌套在 conversations 下，体现「消
    分页参数：
        limit: 每页条数
        offset: 偏移量
    """
    # 先按会话过滤，再统计总数
    base_query = db.query(Message).filter(Message.conversation_id == conversation_id)
    total = base_query.count()

    # 按时间正序（对话按自然顺序展示，最早的在前）
    items = (
        base_query
        .order_by(Message.created_at.asc())
        .offset(offset)
        .limit(limit)
        .all()
    )

    return ApiResponse.success(
        data=PaginatedData(
            items=items,
            total=total,
            limit=limit,
            offset=offset,
        )
    )

你需要观察什么

1. 不同操作的 HTTP 状态码：创建 201、查询 200、删除 204、不存在 404。这些状态码是 RESTful 的「语义信号」——调用方不用看 body 就知道发生了什么。
3. 分页的 total 字段：创建 3 个会话，然后 limit=2 查询，看返回的 items 是 2 条但 total 是 3。这个 total 让前端知道「还有下一页」。
4. exclude_unset=True 的作用：PATCH 更新时，如果客户端只传 {"title": "新标题"}，model_dump(exclude_unset=True) 只返回 {"title": ...}。如果去掉这个参数，会把所有字段（包括没传的 None）都写进去，覆盖掉原值。
